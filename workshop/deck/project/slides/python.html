from statistics import NormalDist

def single_proportion(p, d, alpha=0.05):
    z = NormalDist().inv_cdf(1 - alpha / 2)
    return z**2 * p * (1 - p) / d**2

single_proportion(0.5, 0.05)  # 384.1 → 385
